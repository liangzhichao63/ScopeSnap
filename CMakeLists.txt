cmake_minimum_required(VERSION 3.21)
project(ScopeCapture VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network Test)
add_library(scope_backend STATIC src/scopeworker.cpp src/scopeworker.h src/imagecanvas.cpp src/imagecanvas.h)
target_link_libraries(scope_backend PUBLIC Qt6::Widgets Qt6::Network)
target_include_directories(scope_backend PUBLIC src)
add_executable(ScopeCapture WIN32 src/main.cpp src/mainwindow.cpp src/mainwindow.h resources/images.qrc)
if(WIN32)
  target_sources(ScopeCapture PRIVATE resources/version.rc)
endif()
target_link_libraries(ScopeCapture PRIVATE scope_backend Qt6::Widgets Qt6::Network)
target_compile_definitions(ScopeCapture PRIVATE APP_VERSION="${PROJECT_VERSION}")
include(CTest)
if(BUILD_TESTING)
  add_executable(scope_tests tests/tst_scope.cpp src/mainwindow.cpp src/mainwindow.h resources/images.qrc)
  target_compile_definitions(scope_tests PRIVATE APP_VERSION="${PROJECT_VERSION}")
  target_link_libraries(scope_tests PRIVATE scope_backend Qt6::Test Qt6::Widgets Qt6::Network)
  add_test(NAME scope_tests COMMAND scope_tests)
endif()
